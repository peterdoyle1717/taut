    42	/-- The four triangular faces of a tetrahedron `t` (the card-3 subsets). -/
    43	abbrev tetFaces (t : Finset V) : Finset (Finset V) := t.powersetCard 3

    45	/-- One gluing step of a shelling: tet `t` is glued onto a ball whose
    46	boundary 2-sphere is `B`, producing the ball with boundary `B'`.
    47
    48	* `card4` — `t` is a tetrahedron (4 vertices);
    49	* `shared` — `t` meets the current boundary in **one or two** of its faces
    50	  (type 1 or 2; type 3 is excluded, as it would create an interior vertex);
    51	* `newBdry` — the new boundary is the symmetric difference of `B` and the
    52	  faces of `t`: shared faces become interior, the rest are exposed;
    53	* `sphere` — the new boundary is again a combinatorial 2-sphere (the
    54	  invariant baked into the certificate).

    56	structure GlueStep (t : Finset V) (B B' : Finset (Finset V)) : Prop where
    57	  card4 : t.card = 4
    58	  shared : (tetFaces t ∩ B).card = 1 ∨ (tetFaces t ∩ B).card = 2
    59	  newBdry : B' = (B \ tetFaces t) ∪ (tetFaces t \ B)

    65	  | B₀, [],     B => B = B₀
