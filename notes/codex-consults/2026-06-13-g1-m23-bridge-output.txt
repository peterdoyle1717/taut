   155	/-- `∂₁c` at a vertex `x` lying on exactly the two support-edges `e₁, e₂` equals
   156	`c e₁ + c e₂`. -/
   157	lemma bd1_eval_two {σ : Finset (Finset V)} {c : C1 σ} {x : V} (hx : x ∈ vertsOf σ)
   158	    {e₁ e₂ : edgesOf σ} (hne : e₁ ≠ e₂) (hx1 : x ∈ (e₁ : Finset V)) (hx2 : x ∈ (e₂ : Finset V))

   164	  · rw [Finset.sum_pair hne, if_pos hx1, if_pos hx2, one_mul, one_mul]
   165	  · -- terms off {e₁,e₂} vanish
   166	    by_cases hxe : x ∈ (e : Finset V)
   167	    · by_cases hce : c e = 0
   168	      · rw [hce, mul_zero]
   169	      · rcases honly e hxe hce with rfl | rfl <;> simp_all
   170	    · rw [if_neg hxe, zero_mul]

   172	/-- A 1-cycle supported on the edges of `γ` (a triangle whose 2-subsets are
   173	edges) is `0` or the whole γ-cycle: `∂₁ = 0` forces the three coefficients
   174	equal. -/
   175	lemma gammaCycle_dichotomy {σ : Finset (Finset V)} {γ : Finset V} (hγ3 : γ.card = 3)
   176	    (hγe : γ.powersetCard 2 ⊆ edgesOf σ) {c : C1 σ} (hc : bd1 σ c = 0)
   177	    (hsupp : ∀ e : edgesOf σ, c e ≠ 0 → (e : Finset V) ⊆ γ) :
   178	    c = 0 ∨ c = gammaChain σ γ := by
   179	  classical
   180	  obtain ⟨a, b, d, hab, had, hbd, hγ⟩ := Finset.card_eq_three.mp hγ3
   181	  have mk : ∀ {p q : V}, p ≠ q → ({p, q} : Finset V) ⊆ γ → ({p, q} : Finset V) ∈ edgesOf σ :=
   182	    fun hpq hsub => hγe (Finset.mem_powersetCard.mpr ⟨hsub, Finset.card_pair hpq⟩)
   183	  have sab : ({a, b} : Finset V) ⊆ γ := by
   184	    rw [hγ]; intro z hz; simp only [Finset.mem_insert, Finset.mem_singleton] at hz ⊢; tauto
   185	  have sad : ({a, d} : Finset V) ⊆ γ := by
   186	    rw [hγ]; intro z hz; simp only [Finset.mem_insert, Finset.mem_singleton] at hz ⊢; tauto
   187	  have sbd : ({b, d} : Finset V) ⊆ γ := by
   188	    rw [hγ]; intro z hz; simp only [Finset.mem_insert, Finset.mem_singleton] at hz ⊢; tauto
   189	  set eab : edgesOf σ := ⟨{a, b}, mk hab sab⟩ with heab
   190	  set ead : edgesOf σ := ⟨{a, d}, mk had sad⟩ with head
   191	  set ebd : edgesOf σ := ⟨{b, d}, mk hbd sbd⟩ with hebd
   192	  -- every γ-edge is one of the three
   193	  have tri : ∀ e : edgesOf σ, (e : Finset V) ⊆ γ → e = eab ∨ e = ead ∨ e = ebd := by
   194	    intro e hsub
   195	    obtain ⟨p, q, hpq, hpqe⟩ := Finset.card_eq_two.mp (card_of_mem_edgesOf e.2)
   196	    rw [hγ] at hsub
   197	    have hp : p ∈ ({a, b, d} : Finset V) := hsub (hpqe ▸ by simp)
   198	    have hq : q ∈ ({a, b, d} : Finset V) := hsub (hpqe ▸ by simp)
   199	    simp only [Finset.mem_insert, Finset.mem_singleton] at hp hq
   200	    have key : (e : Finset V) = {a, b} ∨ (e : Finset V) = {a, d} ∨ (e : Finset V) = {b, d} := by
   201	      rw [hpqe]
   202	      rcases hp with rfl | rfl | rfl <;> rcases hq with rfl | rfl | rfl <;>
   203	        first
   204	          | exact absurd rfl hpq
   205	          | exact Or.inl rfl
   206	          | exact Or.inl (Finset.pair_comm _ _)
   207	          | exact Or.inr (Or.inl rfl)
   208	          | exact Or.inr (Or.inl (Finset.pair_comm _ _))
   209	          | exact Or.inr (Or.inr rfl)
   210	          | exact Or.inr (Or.inr (Finset.pair_comm _ _))
   211	    rcases key with h | h | h
   212	    · exact Or.inl (Subtype.ext h)
   213	    · exact Or.inr (Or.inl (Subtype.ext h))
   214	    · exact Or.inr (Or.inr (Subtype.ext h))
   215	  have ha : a ∈ vertsOf σ := edge_mem_vertsOf (mk hab sab) (by simp)
   216	  have hb : b ∈ vertsOf σ := edge_mem_vertsOf (mk hab sab) (by simp)
   217	  have hd : d ∈ vertsOf σ := edge_mem_vertsOf (mk had sad) (by simp)
   218	  have ne_ab_ad : eab ≠ ead := by
   219	    rw [heab, head, ne_eq, Subtype.mk.injEq]; intro h
   220	    have : b ∈ ({a, d} : Finset V) := h ▸ by simp
   221	    simp only [Finset.mem_insert, Finset.mem_singleton] at this; tauto
   222	  have ne_ab_bd : eab ≠ ebd := by
   223	    rw [heab, hebd, ne_eq, Subtype.mk.injEq]; intro h
   224	    have : a ∈ ({b, d} : Finset V) := h ▸ by simp
   225	    simp only [Finset.mem_insert, Finset.mem_singleton] at this; tauto
   226	  have ne_ad_bd : ead ≠ ebd := by
   227	    rw [head, hebd, ne_eq, Subtype.mk.injEq]; intro h
   228	    have : a ∈ ({b, d} : Finset V) := h ▸ by simp
   229	    simp only [Finset.mem_insert, Finset.mem_singleton] at this; tauto
   230	  -- the three vertex equations
   231	  have hva : c eab + c ead = 0 := by
   232	    have honly : ∀ e : edgesOf σ, a ∈ (e : Finset V) → c e ≠ 0 → e = eab ∨ e = ead := by
   233	      intro e hxe hce
   234	      rcases hone : tri e (hsupp e hce) with h | h | h
   235	      · exact Or.inl h
   236	      · exact Or.inr h
   237	      · exact absurd (h ▸ hxe) (by rw [hebd]; simp only [Finset.mem_insert, Finset.mem_singleton]; tauto)
   238	    rw [← bd1_eval_two ha ne_ab_ad (by rw [heab]; simp) (by rw [head]; simp) honly, hc]; rfl
   239	  have hvb : c eab + c ebd = 0 := by
   240	    have honly : ∀ e : edgesOf σ, b ∈ (e : Finset V) → c e ≠ 0 → e = eab ∨ e = ebd := by
   241	      intro e hxe hce
   242	      rcases tri e (hsupp e hce) with h | h | h
   243	      · exact Or.inl h
   244	      · exact absurd (h ▸ hxe) (by rw [head]; simp only [Finset.mem_insert, Finset.mem_singleton]; tauto)
   245	      · exact Or.inr h
   246	    rw [← bd1_eval_two hb ne_ab_bd (by rw [heab]; simp) (by rw [hebd]; simp) honly, hc]; rfl
   247	  -- so the three coefficients are equal
   248	  have c2 : ∀ x y : ZMod 2, x + y = 0 → x = y := by decide
   249	  have e1 : c ead = c eab := (c2 _ _ hva).symm
   250	  have e2 : c ebd = c eab := (c2 _ _ hvb).symm
   251	  -- c = (c eab) • gammaChain
   252	  have hrep : c = (c eab) • gammaChain σ γ := by
   253	    funext e
   254	    simp only [Pi.smul_apply, gammaChain, smul_eq_mul]
   255	    by_cases hsub : (e : Finset V) ⊆ γ
   256	    · rw [if_pos hsub, mul_one]
   257	      rcases tri e hsub with h | h | h
   258	      · rw [h]
   259	      · rw [h, e1]
   260	      · rw [h, e2]
   261	    · rw [if_neg hsub, mul_zero]
   262	      by_contra hce
   263	      exact hsub (hsupp e hce)
   264	  -- conclude on the value c eab ∈ {0,1}
   265	  rcases (by decide : ∀ x : ZMod 2, x = 0 ∨ x = 1) (c eab) with h0 | h1
   266	  · left; rw [hrep, h0, zero_smul]
   267	  · right; rw [hrep, h1, one_smul]
